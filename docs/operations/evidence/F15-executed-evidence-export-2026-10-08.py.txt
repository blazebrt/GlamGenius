def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False).encode()).hexdigest()

def safe_manifest(manifest):
    result = {key: manifest[key] for key in ('server_version','database','schema_context','alembic_heads','public_tables','public_table_counts','auth_users','storage_buckets','storage_objects','vault_rows','external_integrations','off_schema_present','later_label_report_resources_present','extensions','sorted_primary_id_sha256')}
    for key in ('public_indexes','public_constraints'):
        result[key] = {'count':len(manifest[key]),'canonical_json_sha256':digest(manifest[key])}
    return result
